-- ─── RANDOM PLAYER FUNCTION ───────────────────────────────
local function getRandomPlayer()
    local validPlayers = {}
    
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LP and plr.Character then
            local vHrp = plr.Character:FindFirstChild("HumanoidRootPart")
            local vHum = plr.Character:FindFirstChildOfClass("Humanoid")
            if vHrp and vHum and vHum.Health > 0 then
                table.insert(validPlayers, plr)
            end
        end
    end
    
    if #validPlayers > 0 then
        return validPlayers[math.random(1, #validPlayers)]
    end
    return nil
end

-- ─── BUTTON LOGIC ─────────────────────────────────────────
followBtn.MouseButton1Click:Connect(function()
    if activeMode == "Follow" then
        activeTarget = nil
        activeMode = nil
        followBtn.Text = "FOLLOW RANDOM"
        followBtn.BackgroundColor3 = Color3.fromRGB(30, 30, 35)
        statusLbl.Text = "Target: None"
    else
        local target = getRandomPlayer()
        if target then
            activeTarget = target
            activeMode = "Follow"
            followBtn.Text = "STOP FOLLOWING"
            followBtn.BackgroundColor3 = Color3.fromRGB(0, 150, 0)
            orbitBtn.Text = "ORBIT RANDOM"
            orbitBtn.BackgroundColor3 = Color3.fromRGB(30, 30, 35)
            statusLbl.Text = "Target: " .. target.Name
        else
            statusLbl.Text = "No players found!"
        end
    end
end)

orbitBtn.MouseButton1Click:Connect(function()
    if activeMode == "Orbit" then
        activeTarget = nil
        activeMode = nil
        orbitBtn.Text = "ORBIT RANDOM"
        orbitBtn.BackgroundColor3 = Color3.fromRGB(30, 30, 35)
        statusLbl.Text = "Target: None"
    else
        local target = getRandomPlayer()
        if target then
            activeTarget = target
            activeMode = "Orbit"
            orbitAngle = 0
            orbitBtn.Text = "STOP ORBITING"
            orbitBtn.BackgroundColor3 = Color3.fromRGB(0, 150, 0)
            followBtn.Text = "FOLLOW RANDOM"
            followBtn.BackgroundColor3 = Color3.fromRGB(30, 30, 35)

        else

        end
    end
end)

-- ─── MAIN LOOP ────────────────────────────────────────────
RunService.Heartbeat:Connect(function()
    if not activeTarget or not activeMode then return end
    
    local myChar = LP.Character
    local myHrp = myChar and myChar:FindFirstChild("HumanoidRootPart")
    if not myHrp then return end
    
    local targetChar = activeTarget.Character
    local targetHrp = targetChar and targetChar:FindFirstChild("HumanoidRootPart")
    
    -- If target dies or leaves, pick a new random one automatically
    if not targetHrp or targetChar.Humanoid.Health <= 0 then
        local newTarget = getRandomPlayer()
        if newTarget then
            activeTarget = newTarget
            statusLbl.Text = "Target: " .. newTarget.Name .. " (Switched)"
        else
            activeTarget = nil
            activeMode = nil
            followBtn.Text = "FOLLOW RANDOM"
            followBtn.BackgroundColor3 = Color3.fromRGB(30, 30, 35)
            orbitBtn.Text = "ORBIT RANDOM"
            orbitBtn.BackgroundColor3 = Color3.fromRGB(30, 30, 35)
            statusLbl.Text = "Target: Lost"
            return
        end
        return
    end

    if activeMode == "Follow" then
        myHrp.CFrame = targetHrp.CFrame
    elseif activeMode == "Orbit" then
        orbitAngle = orbitAngle + math.rad(15)
        local radius = 6
        local offset = Vector3.new(math.cos(orbitAngle) * radius, 0, math.sin(orbitAngle) * radius)
        myHrp.CFrame = CFrame.new(targetHrp.Position + offset)
    end
end)

-- ─── DRAG LOGIC ───────────────────────────────────────────
local dragging, dragStart, startPos
topBar.InputBegan:Connect(function(i)
    if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
        dragging = true; dragStart = i.Position; startPos = main.Position
    end
end)
UIS.InputEnded:Connect(function(i)
    if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then dragging = false end
end)
UIS.InputChanged:Connect(function(i)
    if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
        local d = i.Position - dragStart
        main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
    end
end)
